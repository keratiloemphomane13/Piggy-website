    <script type="module">
        import { initializeApp } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-app.js";
        import { getAuth, onAuthStateChanged, signInAnonymously, signOut } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-auth.js";
        import { getFirestore, collection, onSnapshot, addDoc, doc, updateDoc, deleteDoc } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-firestore.js";

        document.addEventListener('DOMContentLoaded', () => {
            // --- IMPORTANT: PASTE YOUR FIREBASE CONFIGURATION HERE ---
            // For Firebase JS SDK v7.20.0 and later, measurementId is optional
const firebaseConfig = {
  apiKey: "AIzaSyDDCx_z0_i3ZM5t2rbkaVj_u8mc7JFYxwQ",
  authDomain: "piggy-ai.firebaseapp.com",
  projectId: "piggy-ai",
  storageBucket: "piggy-ai.firebasestorage.app",
  messagingSenderId: "1054492593461",
  appId: "1:1054492593461:web:7c3fd5297bd7ce4386507c",
  measurementId: "G-VQT8XE9DF3"
};
            // ---------------------------------------------------------

            // --- FIX: Check for placeholder API keys ---
            if (firebaseConfig.apiKey === "YOUR_API_KEY") {
                alert("Welcome to Piggy! To connect to your database, you need to add your Firebase API keys. Please follow the deployment guide to get your keys from the Firebase website and paste them into the code where it says 'YOUR_API_KEY'.");
                const loadingState = document.getElementById('loading-state');
                if(loadingState) loadingState.textContent = "Configuration needed. Please add your Firebase API keys to the code.";
                return; // Stop the script if keys are missing
            }

            // --- Initialize Firebase ---
            let db, auth, goalsCollectionRef, currentUser, unsubscribeGoals;
            try {
                const app = initializeApp(firebaseConfig);
                auth = getAuth(app);
                db = getFirestore(app);
            } catch (e) {
                console.error("Firebase initialization failed:", e);
                alert("Could not connect to the database. Please check your Firebase configuration.");
                return;
            }
            
            // --- State ---
            let goals = [];
            let activeGoalId = null;

            // --- Elements ---
            const goalsContainer = document.getElementById('goals-container');
            const addGoalBtn = document.getElementById('add-goal-btn');
            const goalModal = document.getElementById('goal-modal');
            const manageGoalModal = document.getElementById('manage-goal-modal');
            const goalForm = document.getElementById('goal-form');
            const addFundsForm = document.getElementById('add-funds-form');
            const modalTitle = document.getElementById('modal-title');
            const manageModalTitle = document.getElementById('manage-modal-title');
            const editGoalBtn = document.getElementById('edit-goal-btn');
            const deleteGoalBtn = document.getElementById('delete-goal-btn');
            const loadingState = document.getElementById('loading-state');
            const welcomeMessage = document.getElementById('welcome-message');
            const signOutBtn = document.getElementById('sign-out-btn');

            // --- Authentication ---
            onAuthStateChanged(auth, user => {
                if (user) {
                    currentUser = user;
                    welcomeMessage.textContent = `My Savings Dashboard`;
                    signOutBtn.classList.remove('hidden');
                    addGoalBtn.disabled = false;
                    addGoalBtn.classList.remove('opacity-50', 'cursor-not-allowed');
                    goalsCollectionRef = collection(db, "users", currentUser.uid, "goals");
                    listenForGoals();
                } else {
                    currentUser = null;
                    welcomeMessage.textContent = 'Signing in...';
                    if (unsubscribeGoals) unsubscribeGoals();
                    goals = [];
                    renderGoals();
                    signOutBtn.classList.add('hidden');
                    addGoalBtn.disabled = true;
                    addGoalBtn.classList.add('opacity-50', 'cursor-not-allowed');
                    signInAnonymously(auth).catch(error => {
                        console.error("Anonymous sign-in failed:", error);
                        loadingState.textContent = "Sign-in failed. Please check your API keys and Firestore rules.";
                    });
                }
            });

            // --- Database Functions ---
            function listenForGoals() {
                if (unsubscribeGoals) unsubscribeGoals();
                unsubscribeGoals = onSnapshot(goalsCollectionRef, (snapshot) => {
                    goals = [];
                    snapshot.forEach((doc) => {
                        goals.push({ ...doc.data(), id: doc.id });
                    });
                    renderGoals();
                }, (error) => {
                    console.error("Error listening for goals:", error);
                    loadingState.textContent = "Error loading goals. Check Firestore rules.";
                });
            }

            // --- Rendering ---
            function renderGoals() {
                if (loadingState) loadingState.classList.add('hidden');
                goalsContainer.innerHTML = '';
                if (goals.length === 0) {
                    goalsContainer.innerHTML = `<p class="text-center text-gray-500 col-span-full">You haven't added any goals yet. Click the button below to start!</p>`;
                } else {
                    goals.forEach(goal => {
                        const progress = (goal.target > 0) ? (goal.current / goal.target) * 100 : 0;
                        const card = document.createElement('div');
                        card.className = 'p-6 bg-white rounded-lg shadow-md';
                        card.innerHTML = `
                            <div class="flex items-center justify-between">
                                <h3 class="text-lg font-semibold text-gray-800">${goal.name}</h3>
                            </div>
                            <p class="mt-2 text-sm text-gray-600">$${goal.current.toFixed(2)} / $${goal.target.toFixed(2)}</p>
                            <div class="w-full h-2 mt-2 bg-gray-200 rounded-full">
                                <div class="h-full bg-green-500 rounded-full" style="width: ${progress}%;"></div>
                            </div>
                            <button data-id="${goal.id}" class="manage-goal-btn w-full mt-4 px-4 py-2 text-sm font-medium text-pink-600 bg-pink-100 rounded-md hover:bg-pink-200">Manage Goal</button>
                        `;
                        goalsContainer.appendChild(card);
                    });
                }
            }

            // --- Modal Logic ---
            function openModal(modal) {
                modal.classList.remove('hidden');
                modal.classList.add('flex');
            }

            function closeModal(modal) {
                modal.classList.add('hidden');
                modal.classList.remove('flex');
            }

            // --- Event Handlers ---
            addGoalBtn.addEventListener('click', () => {
                activeGoalId = null;
                modalTitle.textContent = 'Create a New Savings Goal';
                goalForm.reset();
                document.getElementById('current-amount').disabled = false;
                openModal(goalModal);
            });

            document.querySelectorAll('.close-modal-btn, .cancel-btn').forEach(btn => {
                btn.addEventListener('click', () => {
                    closeModal(goalModal);
                    closeModal(manageGoalModal);
                });
            });

            goalsContainer.addEventListener('click', (e) => {
                if (e.target.classList.contains('manage-goal-btn')) {
                    activeGoalId = e.target.dataset.id;
                    const goal = goals.find(g => g.id === activeGoalId);
                    if(goal) {
                        manageModalTitle.textContent = `Manage "${goal.name}"`;
                        openModal(manageGoalModal);
                    }
                }
            });

            goalForm.addEventListener('submit', async (e) => {
                e.preventDefault();
                const name = document.getElementById('goal-name').value;
                const target = parseFloat(document.getElementById('goal-amount').value);
                const current = parseFloat(document.getElementById('current-amount').value) || 0;

                if (activeGoalId) { // Editing
                    const goalRef = doc(db, "users", currentUser.uid, "goals", activeGoalId);
                    await updateDoc(goalRef, { name, target });
                } else { // Adding
                    await addDoc(goalsCollectionRef, { name, current, target });
                }
                closeModal(goalModal);
            });

            editGoalBtn.addEventListener('click', () => {
                 const goal = goals.find(g => g.id === activeGoalId);
                 if(goal) {
                    modalTitle.textContent = `Edit "${goal.name}"`;
                    document.getElementById('goal-name').value = goal.name;
                    document.getElementById('goal-amount').value = goal.target;
                    document.getElementById('current-amount').value = goal.current;
                    document.getElementById('current-amount').disabled = true;
                    closeModal(manageGoalModal);
                    openModal(goalModal);
                 }
            });

            deleteGoalBtn.addEventListener('click', async () => {
                if (confirm('Are you sure you want to delete this goal?')) {
                    const goalRef = doc(db, "users", currentUser.uid, "goals", activeGoalId);
                    await deleteDoc(goalRef);
                    closeModal(manageGoalModal);
                }
            });

            addFundsForm.addEventListener('submit', async (e) => {
                e.preventDefault();
                const amount = parseFloat(document.getElementById('add-funds-amount').value);
                if (!isNaN(amount) && amount > 0) {
                    const goal = goals.find(g => g.id === activeGoalId);
                    if(goal) {
                        const newAmount = goal.current + amount;
                        const goalRef = doc(db, "users", currentUser.uid, "goals", activeGoalId);
                        await updateDoc(goalRef, { current: newAmount });
                        addFundsForm.reset();
                        closeModal(manageGoalModal);
                    }
                }
            });

            signOutBtn.addEventListener('click', () => {
                signOut(auth);
            });
        });
    </script>
